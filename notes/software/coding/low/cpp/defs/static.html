Static vs non static members


    #include 
    #include 

    class Player {
    public:
        std::string name; // Non-static: Each player gets their own name

        // Static data member: Shared by ALL Player objects
        static int totalPlayers;

        Player(std::string p_name) : name(p_name) {
            totalPlayers++; // Increment the shared count whenever a player is created
        }

        ~Player() {
            totalPlayers--; // Decrement when a player is destroyed
        }

        // Static member function: Belongs to the class, not an object
        static void printTotalPlayers() {
            // Can access static members directly, but CANNOT access 'name' 
            // because there is no 'this' pointer.
            std::cout << "Total active players: " << totalPlayers << "\n";
        }
    };

    // Static data members must be defined outside the class in C++
    int Player::totalPlayers = 0;

    int main() {
        Player::printTotalPlayers(); // Called via Class Name (0 active)

        Player player1("Alice");
        Player player2("Bob");

        // Both instances see the exact same static variable value
        std::cout << "Player count from player1: " << player1.totalPlayers << "\n"; // 2
        std::cout << "Player count from player2: " << player2.totalPlayers << "\n"; // 2

        // Preferred way to call static functions (using scope resolution operator ::)
        Player::printTotalPlayers(); // 2 active

        return 0;
    }
